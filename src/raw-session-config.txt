require('dotenv').config();
const session = require('express-session');

const createSession = session({
    secret: process.env.PUBLIC_API_KEY || 'api_key_123', // Note: use your own API key. This is one just sample. Encode your API key on your .env file 
    resave: false,
    saveUninitialized: true,
    cookie: {
        secure:false,
        maxAge: 1000 * 60 * 60 * 20
    }
});

module.exports = createSession;
